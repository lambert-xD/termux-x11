# CMake build for Lorie Wayland unit tests
# Minimal test runner — zero external dependencies

cmake_minimum_required(VERSION 3.22)

set(LORIE_TEST_SOURCES
    test_main.c
    test_framework.c
    test_build.c
    test_compositor.c
    test_surface.c
    test_renderer.c
    test_input.c
    ../compositor.c
    ../output.c
    ../surface.c
    ../renderer.c
    ../input.c
    ../seat.c
    ../keymap.c
    ../protocols/xdg-shell.c
    ../protocols/linux-dmabuf.c
    ../protocols/wl-data-device-manager.c
    ../xwayland.c
    test_xwayland.c
    test_jni.c
    ../wayland-activity.c
    ../main.c
    test_integration.c
    test_shm_texture.c
    test_buffer_stubs.c
)

add_executable(lorie-wayland-tests ${LORIE_TEST_SOURCES})

target_include_directories(lorie-wayland-tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/..
    # Headers generated by wayland.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/../..
    # Headers generated by wayland-protocols.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/../../wayland-protocols
    # Wayland core headers
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../wayland/wayland/src
    # Pixman headers
    ${CMAKE_CURRENT_SOURCE_DIR}/../../../pixman/pixman
    ${CMAKE_CURRENT_BINARY_DIR}/../../../pixman/pixman
)

target_link_libraries(lorie-wayland-tests PRIVATE
    android
    log
    wayland-server
    wayland-protocols-generated
    pixman-1
)

set_target_properties(lorie-wayland-tests PROPERTIES
    C_STANDARD 11
    C_STANDARD_REQUIRED ON
)

# Ensure protocol headers are generated before test compilation
add_dependencies(lorie-wayland-tests wayland-protocols-generated)

# Register with CTest
enable_testing()
add_test(NAME lorie-wayland-unit COMMAND lorie-wayland-tests)
